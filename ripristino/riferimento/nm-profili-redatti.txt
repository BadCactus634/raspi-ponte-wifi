--- /etc/NetworkManager/system-connections/ponte-lan.nmconnection
[connection]
id=ponte-lan
uuid=<uuid>
type=ethernet
autoconnect-priority=100
interface-name=eth0

[ethernet]

[ipv4]
address1=10.42.0.1/24

[ipv6]
addr-gen-mode=default
method=disabled

[proxy]

type=wifi
autoconnect-priority=100
autoconnect-retries=0
interface-name=wlwan

[ipv6]

[proxy]
